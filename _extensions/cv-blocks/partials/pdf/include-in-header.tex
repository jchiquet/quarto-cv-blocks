% What's expressible as native Quarto/pandoc PDF metadata lives in
% _extension.yml instead (documentclass, classoption, geometry,
% colorlinks/linkcolor/urlcolor/citecolor). Only what has no YAML
% equivalent stays as raw LaTeX here.

% fonts.latex is a no-op override (see partials/pdf/fonts.latex), so
% fontenc/inputenc need loading explicitly.
\usepackage[utf8]{inputenc}
\usepackage[T1]{fontenc}

% Quarto's own template already turns the document's `lang:` into babel's
% main language (documentclass option + \usepackage{babel}), so babel is
% already active here. \ifenglish exposes that choice to consuming
% documents' own raw LaTeX (e.g. personal-macros.tex's EN/FR vocabulary)
% via babel's \iflanguage.
\newif\ifenglish
\iflanguage{english}{\englishtrue}{\englishfalse}

% Header/footer: no native Quarto equivalent.
\usepackage{fancyhdr}
\pagestyle{fancy}
\fancyhf{}
\fancyhead[R]{}
\fancyfoot[RO,LE]{\thepage}
\fancyfoot[LO,RE]{\footnotesize\sc\footer}
\renewcommand{\headrulewidth}{0pt}
\renewcommand{\footrulewidth}{0pt}

% Section titles: package load only -- \titleformat calls are in
% before-body.tex, alongside the colors they use.
\usepackage[explicit]{titlesec}

% array: needed for \multblock's custom R{<width>} column type. Also
% loaded by Quarto's own tables.tex, but load it explicitly so \multblock
% still works if a document sets `tables: false`.
\usepackage{array}

\usepackage{url}
\usepackage{eurosym}
\usepackage{fontawesome}

% float: needed for the [H] placement specifier, so a captioned figure
% stays exactly where it's placed instead of drifting -- see the README,
% "Executable code". Not needed for tables: pandoc emits them as
% `longtable`, which isn't a float.
\usepackage{float}

% Aliases for custom @Type names from the old bibtex/mycv.bst pipeline
% that biblatex doesn't know (biber would otherwise error with a
% misleading "found {...}, expected ="). Harmless if unused.
\DeclareBibliographyAlias{hdrthesis}{phdthesis}
\DeclareBibliographyAlias{invited}{inproceedings}
\DeclareBibliographyAlias{school}{unpublished}
\DeclareBibliographyAlias{seminaire}{unpublished}
\DeclareBibliographyAlias{soft}{misc}
\DeclareBibliographyAlias{preprint}{unpublished}

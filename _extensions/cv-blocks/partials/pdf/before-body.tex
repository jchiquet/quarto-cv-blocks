% Overrides Quarto's before-body.tex (a conditional \maketitle) -- \title
% is repurposed below, so \maketitle wouldn't work here anyway.

% compact drops the running header/footer for a one-page document.
% Independent of `details` (content selection, not page count) -- a
% dossier scientifique can have details:false and still span pages, so it
% keeps the fancy pagestyle from include-in-header.tex by default.
$if(compact)$
\pagestyle{empty}
$endif$

% Quarto's template loads parskip (or a nonzero fallback), adding space
% between \multblock/\block entries (each its own paragraph). Reset to
% zero for the tight spacing the original cv.cls had.
\setlength{\parskip}{0pt}

% \nom/\title come from Quarto's native author/title metadata -- see the
% README, "Usage". \title is article.cls's own command (feeds \maketitle,
% never called here), repurposed via \renewcommand as plain text, reused
% in the footer below and freely in a document's own raw LaTeX.
\renewcommand{\title}{$title$}
\def\nom{$for(authors)$$it.name.literal$$sep$ \and $endfor$}

% Used by the fancyhdr footer in include-in-header.tex.
\newcommand{\footer}{\nom\ -- \title}

% Colors for linkcolor/urlcolor/citecolor and \titleformat below --
% defining them here (after \begin{document}) is fine since hyperref/xcolor
% resolve color names lazily.
\definecolor{mblue}{rgb}{.204,.353,.541}
\colorlet{mred}{red!80!black}

% Plain markdown itemize lists (e.g. a `.qmd`'s own bullet content,
% outside any \multblock/\block) get an mred en-dash instead of LaTeX's
% default bullet/en-dash/asterisk nesting, to match the rest of the
% document's understated, rule-and-color-driven look rather than
% standing out as a generic Markdown list.
\setlist[itemize]{label=\color{mred}{\textendash}}

% Markdown definition lists (`Term\n:   Definition`) get a fixed-width,
% right-aligned italic label column instead of LaTeX's default inline
% bold term -- an aligned "label: text" summary without falling back to
% \multblock's raw-LaTeX table.
\setlist[description]{leftmargin=6.5em, labelwidth=6em, labelindent=0pt, align=right, font=\normalfont\itshape}

% Small caps via URW Avant Garde instead of LaTeX's faked small caps.
% \fontencoding{T1} is needed under LuaLaTeX (defaults to TU); harmless
% under pdflatex, where T1 is already active.
\renewcommand{\sc}{\fontencoding{T1}\fontfamily{pag}\fontseries{m}\fontshape{sc}\selectfont}
\renewcommand{\textsc}[1]{{\sc#1}}

% Quarto sets shift-heading-level-by: -1 whenever `title:` is set, so `##`
% is a document's top heading level (\section) -- \part only applies to a
% `#` heading, which none of the examples use, but stays defined for
% consistency if a document ever does.
\titleformat{\part}{\sc\huge\color{mred}}{}{0pt}{\noindent\hrulefill~#1~\hrulefill}
\titleformat{\section}{\sc\huge\color{mred}}{}{0pt}{\noindent\hrulefill~#1~\hrulefill}
\titleformat{\subsection}{\sc\Large\color{mred}}{}{0pt}{\noindent#1~\hrulefill}
\titleformat{\subsubsection}{\sc\large}{}{0pt}{\noindent#1}
\titleformat{\paragraph}[runin]{\sc}{}{}{#1}[]
\titleformat{\subparagraph}[block]{\sc}{}{}{\noindent#1}[]
\titlespacing{\subparagraph}{0pt}{0pt}{0pt}

% \multblock/\block: the two primitives cv-patterns.lua's raw LaTeX
% output is built from -- see the README ("How it works").
\newcolumntype{R}[1]{>{\raggedleft\let\newline\\\arraybackslash\hspace{0pt}}m{#1}}

\newcommand*{\multblock}[3]{%
  \par\noindent\begin{tabular}[t]{@{}R{.15\linewidth}|p{.8\linewidth}}
    \hfill{\sf#1} & {\sc#2} \\\relax
    \readBlock#3\relax\relax
  \end{tabular}\medskip
}

\newcommand*{\block}[2]{%
  \par\noindent\begin{tabular}{@{}p{.15\linewidth}|p{.8\linewidth}}
    {\hfill\sf#1} & {\sc#2} \\
  \end{tabular}\medskip
}

\newcommand{\readBlock}[2]{%
  \ifx\relax#1\empty
  \else
  \hfill{\em#1} & #2\\
  \expandafter\readBlock
  \fi
}

% TOC entries down to \subsubsection (used by frontpage below).
\setcounter{tocdepth}{3}

% Optional title/cover page (format.cv-blocks-pdf.frontpage: true)
$if(frontpage)$
\thispagestyle{plain}
\begin{center}
  \huge{\textsc{\nom}} \\
  \vspace{\baselineskip}
  \Large{\textsc{\title\\}}
\end{center}
\vspace{2.5\baselineskip}
\begin{center}{Last update:\ \today}\end{center}
\tableofcontents
\newpage
$endif$
